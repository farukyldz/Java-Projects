public class Main {
    public static void main(String[] args) {
        int temp, units, tens, hundreds, sum;

        // Loop through all 3-digit numbers
        for (int i = 100; i <= 999; i++) {
            temp = i;

            // Extract the digits
            units = temp % 10;      // Digit in the 1s place
            temp = temp / 10;

            tens = temp % 10;       // Digit in the 10s place
            temp = temp / 10;

            hundreds = temp % 10;   // Digit in the 100s place

            // Calculate the sum of the cubes of its digits
            sum = (units * units * units) + (tens * tens * tens) + (hundreds * hundreds * hundreds);

            // Check if the number is an Armstrong number
            if (i == sum) {
                System.out.println(i + " is an Armstrong number.");
            }
        }
    }
}
