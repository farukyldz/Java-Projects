
public class Main {
    public static void main(String[] args) {
       // The switch-case structure allows us to write multiple if-else statements
        // and makes the code easier to understand.
        Scanner scan=new Scanner(System.in);
        System.out.print("Please enter a number:");
        int num= scan.nextInt();
        switch (num){

            case 1:
                System.out.println("is equal to 1:");
                break;
            case 2:
                System.out.println("is equal to 2:");
                break;
            case 3:
                System.out.println("is equal to 3:");
                break;
            default:
                System.out.println("You have entered an invalid number:");

        }

    }
}
